313def action_evidence_position(evidence, history):
314    """Resolve a trusted or legacy exact span without changing its position."""
315    if evidence.turn_index >= len(history):
316        raise ValueError('M04 action quote outside conversation')
317    turn = history[evidence.turn_index]
318    if turn.role != 'student' or evidence.quote not in turn.text:
319        raise ValueError('M04 action must quote exact student text, not doctor text or paraphrase')
320    start = evidence.source_start if evidence.source_start is not None else turn.text.find(evidence.quote)
321    if turn.text[start:start+len(evidence.quote)] != evidence.quote:
322        raise ValueError('M04 action position does not match its exact source text')
323    return evidence.turn_index, start, start + len(evidence.quote)
